\section{Ridge Regression Analysis}
We estimate $\tb$ via the ridge-regression
\begin{align}
\tbh=\Fb_s(\bth)\quad\text{where}\quad \bth=\arg\min_{\bt'} 0.5\tn{\y-\X\bt'}^2+0.5\la n \tn{\bt'}^2.\label{optim me3}
\end{align}
The residual formulation is given via
\[
\hat{\w}=\arg\min_{\w} 0.5\tn{\X_\z\w_\sigma}^2+0.5\la n \tn{\bt-\w}^2
\]
Observing $0.5\tn{\vb}^2=\max_{\ab} \ab^T\vb-0.5\tn{\ab}^2$, we find
\[
\hat{\w}=\arg\min_{\w_{\sigma}} \max_{\ab} \ab^T\X_\z\w_\sigma-0.5\tn{\ab}^2+0.5\la n \tn{\bt-\w}^2.
\]
Fix $\g\sim\Nn(0,\Iden_n),\h\sim\Nn(0,\Iden_p)$. The associated Gordon's form is given by
\[
\hat{\w}=\arg\min_{\w_{\sigma}} \max_{\ab}\tn{\w_\sigma}\g^T\ab- \tn{\ab}\w_\sigma^T\h-0.5\tn{\ab}^2+0.5\la n \tn{\bt-\w}^2.
\]
Setting $\ab=a\bar{\g}$, this yields 
\[
\min_{\w_{\sigma}} \max_{a\geq 0}a(\tn{\w_\sigma}\tn{\g}- \w_\sigma^T\h)-0.5a^2+0.5\la n \tn{\bt-\w}^2.
\]
Differentiating and setting $a=\tn{\w_\sigma}\tn{\g}- \w_\sigma^T\h$, we obtain \red{Here: need to be more careful. This holds only if $\tn{\w_\sigma}\tn{\g}- \w_\sigma^T\h>0$}
\[
\min_{\w_{\sigma}} \la n \tn{\bt-\w}^2+(\tn{\w_\sigma}\tn{\g}- \w_\sigma^T\h)^2.
\]
Now, set $\w=h\bar{\h}+\gamma \bt$. We obtain
\[
\tn{\w_\sigma}\tn{\g}- \w_\sigma^T\h\approx \sqrt{n}\sqrt{h^2+\alpha\gamma^2+\sigma^2}-h\sqrt{p}.
\]
Secondly, note that
\[
 \tn{\bt-\w}^2=\alpha(1-\gamma)^2+h^2.
\]
Normalizing by $n$ and setting $\pbar=p/n$, we obtain
\begin{align}\label{eq:min_hgamma_reg}
\min_{h,\gamma} \la (\alpha(1-\gamma)^2+h^2)+(\sqrt{h^2+\alpha\gamma^2+\sigma^2}-h\sqrt{\pbar})^2.
\end{align}
Set $T=\sqrt{h^2+\alpha\gamma^2+\sigma^2}$. Differentiating, we find
\begin{align}
&\la \alpha (\gamma-1)+(T-h\sqrt{\pbar})\frac{\alpha\gamma}{T}=0\implies \gamma=\frac{1}{1+\la^{-1}(1-\frac{h\sqrt{\pbar}}{T})}\label{eq:reg1}\\
&\la h+(T-h\sqrt{\pbar})(\frac{h}{T}-\sqrt{\pbar})=0.\label{eq:reg2}
\end{align}


From \eqref{eq:reg1}, we find that 
\begin{align}
T =\Big( \frac{\gamma\sqrt{\pbar}}{\la(\gamma-1)+\gamma}\Big) h
\end{align}
Substitute this in \eqref{eq:reg2} and use the fact that $T>0\rightarrow h\neq 0$, yields:
\begin{align}
\gamma = \frac{1}{2\pbar}\,\Big( -1 - \la + \pbar + \sqrt{(1+\la-\pbar)^2+4\la \pbar} \,\Big) = \frac{1}{2}\,\Big( 1 - \labar -\pbar^{-1} + \sqrt{\Big(1 - \labar-\pbar^{-1} \Big)^2+4\labar} \,\Big)\label{eq:reg_gamma},
\end{align}
where we have also used that optimal $\gamma$ must be non-negative, which is easily seen from \eqref{eq:min_hgamma_reg}. We have also defined,
$$
\labar = \la/\pbar. 
$$
 Note that as $\la\rightarrow0$:
$$
\gamma \stackrel{\la\rightarrow0}{\longrightarrow}  \frac{1}{2\pbar}\Big( \pbar - 1 + \sqrt{(1-\pbar)^2} \,\Big) = \begin{cases}
0 &, \text{ if } \pbar<1,\\
1-\frac{1}{\pbar} &, \text{ if } \pbar>1.
\end{cases}
$$

Next, we can solve \eqref{eq:reg1} to find $h^2$:
\begin{align}\label{eq:h2_reg}
h^2 &= \frac{\alpha\gamma^2+\sigma^2}{\frac{\gamma^2 \pbar}{(\gamma\la+\gamma-\la)^2}-1} = \frac{\alpha\gamma^2+\sigma^2}{\frac{\pbar^{-1}}{\left(\labar+\pbar^{-1}-\frac{\labar}{\gamma}\right)^2}-1}\\
&= \frac{\frac{\alpha}{4}\left( 1 - \labar -\pbar^{-1} + \sqrt{\big(1 - \labar-\pbar^{-1} \Big)^2+4\labar} \,\right)^2 + \sigma^2}{\frac{4\pbar^{-1}}{\left(1+\pbar^{-1} + \labar - \sqrt{(1+\pbar^{-1} + \labar)^2-4\pbar^{-1}} \right)^2}-1}\nn
\end{align}
where to arrive in the last expression we used \eqref{eq:reg_gamma} and the fact that
$$
\labar+\pbar^{-1}-\frac{\labar}{\gamma} = \frac{1}{2}\left(1+\pbar^{-1} + \labar - \sqrt{(1+\pbar^{-1} + \labar)^2-4\pbar^{-1}} \right).
$$
With this and using the fact that $a-\sqrt{a^2-b}<\sqrt{b}$ it can be verified that the denominator in \eqref{eq:h2_reg} is positive. 

From \eqref{eq:h2_reg} note that as $\la\rightarrow 0$
$$
h^2 \stackrel{\la\rightarrow0}{\longrightarrow}  \begin{cases}
\sigma^2\frac{\pbar}{1-\pbar} &, \text{ if } \pbar<1,\\
\sigma^2\frac{1}{\pbar-1} \red{+ \alpha \pbar^{-1}(\pbar^{-1}-1)} &, \text{ if } \pbar>1
\end{cases}
$$
