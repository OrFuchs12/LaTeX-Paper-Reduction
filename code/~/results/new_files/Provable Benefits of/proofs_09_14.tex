\section{Overdetermined analysis}
\begin{lemma}[To be proven rigorously] Set $\pbar=p/n$ and $\sbar=s/n$. Suppose $\pbar<1$ and suppose $\tb^T\bSi\tb=\theta$. Then
\[
\Lc(\bth)=\sigma^2+w^2=\frac{\sigma^2}{1-\pbar}\quad\text{and}\quad \Lc(\tbh)=\frac{1-\pbar+\sbar}{1-\pbar}\sigma^2+\alpha-\theta.
\]
\end{lemma}
\subsection{Analysis}
Since $\pbar<1$, \eqref{optim me} has a unique solution. Set $\wo=\sqrt{\bSi}\w$ where $\w=\bt-\bt'$. Define $\X_\z=[\Xb~\z]$ and $\wo_\sigma=[\wo^T~\sigma]^T$. This leads to the optimization problem
\[
\hat{\wo}=\arg\min_{\wo} \tn{\Xb\wo+\sigma \z}=\arg\min_{\wo} \tn{\X_\z\wo_\sigma}.
\]
Fix $\g\sim\Nn(0,\Iden_p),\h\sim\Nn(0,\Iden_n)$. Note that that $s/p=\sbar/\pbar$. Applying Gordon's Lemma (non-rigorously) and noticing optimal $\wo$ has form $\wo=w\bar{\g}$ where $\bar{\g}=\g/\tn{\g}$, we find
\begin{align}
\min_{\w} \max_{\tn{\ab}\leq 1} \h^T\ab \tn{\wo_{\sigma}}-\tn{\ab}\g^T\wo_{\sigma}&=\min_{\w}  \sqrt{n} \tn{\wo_{\sigma}}-\g^T\wo_{\sigma}\\
&=\min_{\w}  \sqrt{n} \sqrt{w^2+\sigma^2}-w\tn{\g}\\
&\propto\min_{\w}  \sqrt{w^2+\sigma^2}-w\sqrt{\pbar}.
\end{align}
This implies $w=\sqrt{\frac{\pbar}{1-\pbar}}\sigma$ via
\[
\sqrt{\pbar}=\frac{w}{\sqrt{w^2+\sigma^2}}\iff \pbar (w^2+\sigma^2)=w^2\iff w=\sqrt{\frac{\pbar}{1-\pbar}}\sigma
\]
Now, recall that $\bth=\bt+\bSi^{-1/2}\wo$. Since the entries of $\wo$ are normally distributed, we have that
\begin{align}
\sqrt{\bSi}\tbh&=\sqrt{\bSi}\Fb_s(\bth)=\sqrt{\bSi}(\Fb_s(\bt)+\Fb_s(\bSi^{-1/2}\wo))\\
&=\sqrt{\bSi}\Fb_s(\bt)+\Fb_s(\wo)\\
&=\tb+\Fb_s(\wo).
\end{align}
Note that $\tn{\Fb_s(\wo)}=\sqrt{\frac{s}{p}}\sqrt{\frac{\pbar}{1-\pbar}}\sigma=\sqrt{\frac{\sbar}{1-\pbar}}\sigma$. Thus the test error \eqref{test formula} of $\bth$ and $\tbh$ are
\[
\Lc(\bth)=\sigma^2+w^2=\frac{\sigma^2}{1-\pbar}\quad\text{and}\quad \Lc(\tbh)=\frac{\sbar}{1-\pbar}\sigma^2+\alpha-\theta+\sigma^2.
\]
\section{Overparameterized analysis}
Let $\la_i$ be the $i$th diagonal entry of the covariance matrix $\bSi$.
\begin{lemma}[To be proven rigorously] Set $\pbar=p/n$ and $\sbar=s/n$. Then, define the quantities
\begin{align}
&\Lambda\quad\text{is solution of}\quad n=\sum_{i=1}^p\frac{1}{1+(\Lambda\la_i)^{-1}},\\
&\zeta_i=\frac{1}{1+\Lambda \la_i},\\
&\gamma_i=\frac{\Gamma}{n(1+(\Lambda \la_i)^{-1})}\quad\text{where}\quad n(\sigma^2+\sum_{i=1}^n \gamma_i^2+\la_i\zeta_i^2\bt_i^2)=\Gamma^2.
\end{align}
$\bar{\gamma}=\sqrt{p}\gamma,\bar{\Gamma}=\Gamma/\sqrt{p}$
\begin{align}
&\bar{\gamma}_i=\frac{\sqrt{p}\sqrt{p}\bar{\Gamma}}{n(1+(\Lambda \la_i)^{-1})}\quad\text{where}\quad n(\sigma^2+\sum_{i=1}^n \frac{\bar{\gamma}_i^2}{p}+\la_i\zeta_i^2\bt_i^2)=p\bar{\Gamma}^2.
\end{align}
Then, the distribution of $\hat{\bt}$ can be approximated as
\[
\hat{\bt}\sim (1-\zeta)\odot\bt-\bSi^{-1/2}\gamma\odot\g.
\]
Then, the test errors are given by
\begin{align}
&\Lc(\bth)=\sigma^2+\sum_{i=1}^p\gamma_i^2+\la_i\zeta_i^2\bt_i^2\\
&\Lc(\tbh)=\sigma^2+\sum_{i=1}^s(\gamma_i^2+\la_i\zeta_i^2\bt_i^2)+\sum_{i=s+1}^p\la_i\bt_i^2.
\end{align}
\end{lemma}
\subsection{Derivation}
Since $\pbar>1$, we solve the least-squares problem  \eqref{optim me2} subject to the min norm constraint. Using change of variable, this yields
\begin{align}
\min_{\w} \tn{\bt-\w}\quad\text{subject to}\quad \tn{\X\w+\sigma \z}=0.
\end{align}
Noticing $\tn{\X\w+\sigma \z}=\tn{\Xb\sqrt{\bSi}\w+\sigma \z}$, Gordon's Lemma formulation takes the form (non-rigorously)
\begin{align}
\min_{\w} \tn{\bt-\w}\quad\text{subject to}\quad \sqrt{n}\tn{\sqrt{\bSi}\w~{\sigma}}=\g^T\sqrt{\bSi}\w.
\end{align}
Setting $w_i=\frac{\gamma_i}{\sqrt{\la_i}}g_i+\zeta_i\bt_i$, we have $\E[\g^T\sqrt{\bSi}\w]=\E[\sum_{i=1}^p\gamma_ig_i^2]=\sum_{i=1}^p\gamma_i:=\Gamma$. Taking expectations, we can write
\begin{align}
\min_{\gamma_i} \sum_{i=1}^p (1-\zeta_i)^2\bt_i^2+\frac{\gamma_i^2}{\la_i} \quad\text{subject to}\quad n(\sigma^2+\sum_{i=1}^n \gamma_i^2+\la_i\zeta_i^2\bt_i^2)=\Gamma^2.
\end{align}
In Lagrangian form, we obtain
\begin{align}
\min_{\gamma_i} \sum_{i=1}^p (1-\zeta_i)^2\bt_i^2+\frac{\gamma_i^2}{\la_i} +\Lambda [(\sigma^2+\sum_{i=1}^n \gamma_i^2+\la_i\zeta_i^2\bt_i^2)-\frac{\Gamma^2}{n}].
\end{align}
Differentiating with respect to $\gamma_i,\zeta_i$, we obtain the equations
\begin{align}
\frac{\gamma_i}{\la_i}+\Lambda [\gamma_i-\Gamma/n]=0&\iff \gamma_i=\frac{\Gamma}{n(1+(\Lambda\la_i)^{-1})}\\
(\zeta_i-1)\bt_i^2+\Lambda \la_i\bt_i^2\zeta_i=0&\iff \zeta_i=\frac{1}{1+\Lambda \la_i}.
\end{align}
$\Lambda$ has to satisfy the inequality
\begin{align}
n=\sum_{i=1}^p\frac{1}{1+(\Lambda\la_i)^{-1}}\iff \sum_{i=1}^p\gamma_i=\Gamma\sum_{i=1}^p\frac{1}{n(1+(\Lambda\la_i)^{-1})}.
\end{align}
Finally, we pick $\Gamma>0$ and $\gamma_i=\frac{\Gamma}{n(1+(\Lambda \la_i)^{-1})}$ such that the following equality is satisfied
\[
n(\sigma^2+\sum_{i=1}^n \gamma_i^2+\la_i\zeta_i^2\bt_i^2)=\Gamma^2.
\]
Finally, fixing a fresh example $\x=\sqrt{\bSi}\bar{\x}$, the test error is given as follows
\[
\Lc(\bth)=\sigma^2+\tn{\sum_{i=1}^p \bar{x}_i\sqrt{\la_i}(\frac{\gamma_i}{\sqrt{\la_i}}g_i+\zeta_i\bt_i)}^2=\sigma^2+\sum_{i=1}^p\gamma_i^2+\la_i\zeta_i^2\bt_i^2.
\]
To proceed, the error of the sparse model can be calculated as follows.
\[
\Lc(\tbh)=\sigma^2+\tn{\sum_{i=1}^s\bar{x}_i \sqrt{\la_i}(\frac{\gamma_i}{\sqrt{\la_i}}g_i+\zeta_i\bt_i)+\sum_{i=s+1}^p\bar{x}_i\sqrt{\la_i}\bt_i}^2=\sigma^2+\sum_{i=1}^s(\gamma_i^2+\la_i\zeta_i^2\bt_i^2)+\sum_{i=s+1}^p\la_i\bt_i^2.
\]

\subsubsection{Proof stuff}
Fix a scalar $\theta>0$ and define the set $\Theta=\{x\bgl \theta(1-\eps)\leq x\leq \theta(1+\eps)\}$. Consider the optimization
\[
\min_{\tn{\bt}\not\in \Theta}\tn{\X\w}
\]
We would like to assess the properties of the solution $\bth$.
In this case, we solve least squares subject to min norm constraint. Gordon takes the form
\begin{align}
\min_{\bt'} \tn{\bt'}\quad\text{subject to}\quad \sqrt{n}\tn{\w_{\sigma}}=\g^T\w_{\sigma}.
\end{align}
This time $\w$ will choose the form
\[
\w=w\bar{\g}-\gamma \bt.
\]
which yields the scalar problem
\[
\min_{w,\gamma} \alpha^2(1-\gamma)^2+w^2\quad\text{subject to}\quad (w^2+\alpha^2\gamma^2+\sigma^2)=w^2\kappa.
\]
This means
\[
w^2=\frac{\alpha^2\gamma^2+\sigma^2}{\kappa-1}\implies \min_{\gamma} \alpha^2(1-\gamma)^2+\frac{\alpha^2\gamma^2+\sigma^2}{\kappa-1}
\]
This implies
\[
1-\gamma=\frac{\gamma}{\kappa-1}\implies 1=\gamma(1+\frac{1}{\kappa-1})\implies \gamma=\frac{\kappa-1}{\kappa}.
\]
In terms of $w,\gamma$, 
\begin{align}
&\bth=(1-\gamma)\bt+w\bar{\g}=\frac{\bt}{\kappa}+w\bar{\g}\\
&\tbh=(1-\gamma)\tb+w\bar{\g}[1:s]
\end{align}
Thus,
\begin{align}
&\bt-\bth=\bt(1-\frac{1}{\kappa})+\bt[k:]+w\bar{\g}\\
&\bt-\tbh=\bt[1:s](1-\frac{1}{\kappa})+\bt[s:]+w\bar{\g}[1:s]
\end{align}
Secondly, note that
\[
w^2=\frac{\alpha^2\gamma^2+\sigma^2}{\kappa-1}=\frac{\sigma^2}{\kappa-1}+\frac{\alpha^2(\kappa-1)}{\kappa^2}
\]

%\min w\quad\text{subject to}\quad w^2+\sigma^2=\kappa w^2.
%\end{align}

Thus the test error of $\bth$ is
\[
err(\bth)=\sigma^2+w^2+\alpha^2(1-\frac{1}{\kappa})^2=\frac{\kappa \sigma^2}{\kappa-1}+\alpha^2\frac{\kappa-1}{\kappa}
\]
Similarly, the error will be proportionally distributed over $\tbh$ thus test error of $\tbh$ is
\[
err(\tbh)=1-\theta^2+\bar{\sigma}^2+\frac{\xi}{\kappa}w^2+\theta^2(1-\frac{1}{\kappa})^2=1-\theta^2+\bar{\sigma}^2+\frac{\xi (1-\alpha^2+\bar{\sigma}^2)}{\kappa(\kappa-1)}+\frac{\kappa-1}{\kappa^2}(\theta^2(\kappa-1)+\frac{\xi}{\kappa}\alpha^2)
